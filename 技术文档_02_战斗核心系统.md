## 1. 确定性基础：定点数与随机数

> **原则**：帧同步要求所有客户端在同一帧输入下产生完全一致的游戏状态。因此所有战斗逻辑计算必须使用定点数和确定性随机。

### 1.1 定点数（FixedPoint）

```javascript
/**
 * 定点数类（精度 0.001）
 * 存为整数，避免 JS 浮点数在不同平台/V8版本产生微小差异
 * 
 * @example
 * FixedPoint.fromFloat(3.14)   → value=3140
 * FixedPoint.fromInt(5)        → value=5000
 * a.multiply(b)                → (a.value * b.value) / 1000 四舍五入
 */
class FixedPoint {
    /**
     * @param {number} value - 内部整数存储，精度 0.001
     */
    constructor(value) {
        this.value = value;
    }

    /** @param {number} f @returns {FixedPoint} */
    static fromFloat(f) {
        return new FixedPoint(Math.round(f * 1000));
    }

    /** @param {number} n @returns {FixedPoint} */
    static fromInt(n) {
        return new FixedPoint(n * 1000);
    }

    /** @returns {FixedPoint} */
    static zero() { return new FixedPoint(0); }

    /** @returns {FixedPoint} */
    static get ONE() { return FixedPoint.fromInt(1); }

    /** @returns {FixedPoint} */
    static get ZERO() { return FixedPoint.zero(); }

    /** @param {FixedPoint} other @returns {FixedPoint} */
    add(other) { return new FixedPoint(this.value + other.value); }

    /** @param {FixedPoint} other @returns {FixedPoint} */
    subtract(other) { return new FixedPoint(this.value - other.value); }

    /**
     * 定点数乘法：乘积后除以精度再四舍五入
     * @param {FixedPoint} factor
     * @returns {FixedPoint}
     */
    multiply(factor) {
        return new FixedPoint(Math.round((this.value * factor.value) / 1000));
    }

    /**
     * 定点数除法：被除数×精度再除以除数
     * @param {FixedPoint} divisor
     * @returns {FixedPoint}
     */
    divide(divisor) {
        if (divisor.value === 0) return FixedPoint.zero();
        return new FixedPoint(Math.round((this.value * 1000) / divisor.value));
    }

    /** @param {FixedPoint} other @returns {boolean} */
    greaterThan(other) { return this.value > other.value; }

    /** @param {FixedPoint} other @returns {boolean} */
    greaterOrEqual(other) { return this.value >= other.value; }

    /** @param {FixedPoint} other @returns {boolean} */
    lessThan(other) { return this.value < other.value; }

    /** @param {FixedPoint} other @returns {boolean} */
    equals(other) { return this.value === other.value; }

    /** @returns {number} */
    toFloat() { return this.value / 1000; }

    /** @returns {FixedPoint} 深拷贝 */
    clone() { return new FixedPoint(this.value); }
}
```

### 1.2 确定性随机数（SeededRandom）

```javascript
/**
 * Mulberry32 确定性随机数生成器
 * 相同种子在任何平台/任何时候产生完全相同的随机序列
 */
class SeededRandom {
    /**
     * @param {number} seed - 32位整数种子
     */
    constructor(seed) {
        this.seed = seed | 0;
    }

    /**
     * @returns {number} [0, 1) 浮点数
     */
    next() {
        this.seed |= 0;
        this.seed = this.seed + 0x6D2B79F5 | 0;
        let t = Math.imul(this.seed ^ this.seed >>> 15, 1 | this.seed);
        t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
        return ((t ^ t >>> 14) >>> 0) / 4294967296;
    }

    /**
     * @param {number} min
     * @param {number} max
     * @returns {number} [min, max) 随机整数
     */
    nextInt(min, max) {
        return Math.floor(this.next() * (max - min)) + min;
    }

    /**
     * @param {number} min
     * @param {number} max
     * @returns {number} [min, max] 随机整数
     */
    nextIntInclusive(min, max) {
        return Math.floor(this.next() * (max - min + 1)) + min;
    }

    /**
     * 从数组中随机抽取一个元素
     * @param {T[]} arr
     * @returns {T}
     * @template T
     */
    pickOne(arr) {
        return arr[this.nextInt(0, arr.length)];
    }

    /**
     * 从数组中不重复抽取 n 个元素
     * @param {T[]} arr
     * @param {number} n
     * @returns {T[]}
     * @template T
     */
    pickN(arr, n) {
        const result = [];
        const available = [...arr];
        const count = Math.min(n, available.length);
        for (let i = 0; i < count; i++) {
            const idx = this.nextInt(0, available.length);
            result.push(available[idx]);
            available.splice(idx, 1);
        }
        return result;
    }
}
```

### 1.3 随机数种子体系

> 📌 来自需求 §11.13：明确不同系统共享/独立种子的细节

| 维度 | 方案 |
|------|------|
| **对局种子（room_seed）** | 房主客户端生成（或服务端生成），下发给所有队友 |
| **地图派生** | `mapRng = new SeededRandom(hash(room_seed + "map"))` |
| **怪物序列派生** | `monsterRng = new SeededRandom(hash(room_seed + "monster"))` |
| **商店商品派生** | `shopRng = new SeededRandom(hash(room_seed + "shop" + waveNumber))` |
| **骰子结果** | 服务端独立真随机，**不依赖客户端种子**（防预测） |
| **抽卡结果** | 服务端独立随机，含保底计算 |
| **一致性验证** | 每3波服务端计算 `hash(客户端状态)`，不一致则追帧 |
| **种子安全** | room_seed 不在UI暴露 |

## 2. 随机地图生成

### 2.1 地图规格

| 参数 | 值 | 说明 |
|------|-----|------|
| 地图尺寸 | 20×15 格 | 宽20高15 |
| 障碍物密度 | 15%~30% | 随机浮动 |
| 可放置区域 | 路径两侧 2 格 | + 额外随机 5%~10% |
| 装饰物 | 树木/岩石/花草 | 纯装饰，不影响逻辑 |

### 2.2 瓦片类型

```javascript
/** @enum {number} */
const TileType = Object.freeze({
    EMPTY: 0,       // 空地
    PATH: 1,        // 怪物路径
    OBSTACLE: 2,    // 障碍物（不可放置不可通行）
    PLACEABLE: 3,   // 可放置塔
    ENTRANCE: 4,    // 怪物入口
    CRYSTAL: 5,     // 水晶位置
});
```

### 2.3 生成流程

```
Step 1: BSP 二叉空间分割 → 将 20×15 地图递归划分约 8-12 个子区域
Step 2: 各子区域按 15%-30% 随机密度放置障碍物
Step 3: 障碍物过多时自动清理保证连通性
Step 4: A* 算法确保从入口到水晶有可达路径
Step 5: 路径两侧 2 格内标记为 PLACEABLE
Step 6: 额外随机散布 5%-10% PLACEABLE 格（非障碍区域）
Step 7: 放置入口（左侧中部）和水晶（右侧中部）
Step 8: 随机放置装饰物（树木/岩石/花草）
```

### 2.4 核心代码

```javascript
/**
 * @typedef {object} Rect
 * @property {number} x
 * @property {number} y
 * @property {number} w
 * @property {number} h
 */

/**
 * @typedef {object} GridPoint
 * @property {number} x
 * @property {number} y
 */

/**
 * @typedef {object} GameMap
 * @property {number} width          - 20
 * @property {number} height         - 15
 * @property {TileType[][]} tiles    - tiles[y][x]
 * @property {GridPoint[]} path      - 怪物行走路径（A*结果）
 * @property {GridPoint} spawnPoint  - 怪物出生点
 * @property {GridPoint} crystalPoint- 水晶位置
 * @property {number} seed           - 地图种子
 * @property {GridPoint[][]} allPaths - 所有可能的路径（热力图用）
 */

class MapGenerator {
    /**
     * @param {number} mapSeed
     * @param {number} [width=20]
     * @param {number} [height=15]
     * @returns {GameMap}
     */
    generate(mapSeed, width, height) {
        width = width || 20;
        height = height || 15;
        const rng = new SeededRandom(mapSeed);

        // Step 1: BSP 分区
        const regions = this._bspPartition(width, height, 3, rng);

        // Step 2-3: 创建网格+放置障碍物
        const tiles = this._createTileGrid(width, height);
        for (const region of regions) {
            this._placeObstacles(tiles, region, rng);
        }

        // Step 4-6: A*寻路 + 可放置区域
        const spawnPoint = { x: 0, y: Math.floor(height / 2) };
        const crystalPoint = { x: width - 1, y: Math.floor(height / 2) };

        const pathfinder = new AStarPathfinding();
        const path = pathfinder.findPath(tiles, spawnPoint, crystalPoint);
        if (!path) {
            // 路径生成失败，清除部分障碍物后重试
            this._clearBlockingObstacles(tiles, spawnPoint, crystalPoint);
            const retryPath = pathfinder.findPath(tiles, spawnPoint, crystalPoint);
            if (!retryPath) {
                // 极端情况：生成无障碍地图
                return this._generateFallbackMap(width, height, mapSeed);
            }
            return this._finalizeMap(tiles, retryPath, spawnPoint, crystalPoint, mapSeed, rng);
        }

        return this._finalizeMap(tiles, path, spawnPoint, crystalPoint, mapSeed, rng);
    }

    /**
     * @param {number} width
     * @param {number} height
     * @param {number} depth
     * @param {SeededRandom} rng
     * @returns {Rect[]}
     */
    _bspPartition(width, height, depth, rng) {
        /** @type {Rect[]} */
        const result = [{ x: 0, y: 0, w: width, h: height }];
        for (let d = 0; d < depth; d++) {
            const current = [...result];
            for (const rect of current) {
                if (rect.w > 6 || rect.h > 5) {
                    const idx = result.indexOf(rect);
                    result.splice(idx, 1);
                    const [a, b] = this._split(rect, rng);
                    result.push(a, b);
                }
            }
        }
        return result;
    }

    /**
     * @param {Rect} rect
     * @param {SeededRandom} rng
     * @returns {[Rect, Rect]}
     */
    _split(rect, rng) {
        const horizontal = rng.next() < 0.5;
        if (horizontal && rect.h > 4) {
            const splitY = rect.y + rng.nextInt(2, rect.h - 2);
            return [
                { x: rect.x, y: rect.y, w: rect.w, h: splitY - rect.y },
                { x: rect.x, y: splitY, w: rect.w, h: rect.y + rect.h - splitY },
            ];
        } else if (rect.w > 4) {
            const splitX = rect.x + rng.nextInt(2, rect.w - 2);
            return [
                { x: rect.x, y: rect.y, w: splitX - rect.x, h: rect.h },
                { x: splitX, y: rect.y, w: rect.x + rect.w - splitX, h: rect.h },
            ];
        }
        return [rect, rect];
    }

    /** @param {number} w @param {number} h @returns {TileType[][]} */
    _createTileGrid(w, h) {
        const grid = [];
        for (let y = 0; y < h; y++) {
            grid[y] = new Array(w).fill(TileType.EMPTY);
        }
        return grid;
    }

    /**
     * @param {TileType[][]} tiles
     * @param {Rect} region
     * @param {SeededRandom} rng
     */
    _placeObstacles(tiles, region, rng) {
        const density = 0.15 + rng.next() * 0.15;
        for (let y = region.y; y < region.y + region.h; y++) {
            for (let x = region.x; x < region.x + region.w; x++) {
                if (rng.next() < density) {
                    tiles[y][x] = TileType.OBSTACLE;
                }
            }
        }
    }

    /**
     * @param {TileType[][]} tiles
     * @param {GridPoint[]} path
     * @param {GridPoint} spawn
     * @param {GridPoint} crystal
     * @param {number} seed
     * @param {SeededRandom} rng
     * @returns {GameMap}
     */
    _finalizeMap(tiles, path, spawn, crystal, seed, rng) {
        // 标记路径
        for (const p of path) {
            tiles[p.y][p.x] = TileType.PATH;
        }

        // 标记可放置区域（路径两侧2格 + 额外随机）
        const placeableSet = new Set();
        for (const p of path) {
            for (let dy = -2; dy <= 2; dy++) {
                for (let dx = -2; dx <= 2; dx++) {
                    const nx = p.x + dx;
                    const ny = p.y + dy;
                    if (nx >= 0 && nx < tiles[0].length && ny >= 0 && ny < tiles.length) {
                        if (tiles[ny][nx] === TileType.EMPTY) {
                            placeableSet.add(`${nx},${ny}`);
                        }
                    }
                }
            }
        }

        // 额外随机散布 5%-10%
        for (let y = 0; y < tiles.length; y++) {
            for (let x = 0; x < tiles[0].length; x++) {
                if (tiles[y][x] === TileType.EMPTY && !placeableSet.has(`${x},${y}`)) {
                    if (rng.next() < 0.075) { // 7.5%
                        placeableSet.add(`${x},${y}`);
                    }
                }
            }
        }

        for (const key of placeableSet) {
            const [x, y] = key.split(',').map(Number);
            tiles[y][x] = TileType.PLACEABLE;
        }

        // 入口和水晶
        tiles[spawn.y][spawn.x] = TileType.ENTRANCE;
        tiles[crystal.y][crystal.x] = TileType.CRYSTAL;

        return {
            width: tiles[0].length,
            height: tiles.length,
            tiles,
            path,
            spawnPoint: spawn,
            crystalPoint: crystal,
            seed,
        };
    }

    /**
     * 清理堵塞入口/出口的障碍物
     * @param {TileType[][]} tiles
     * @param {GridPoint} spawn
     * @param {GridPoint} crystal
     */
    _clearBlockingObstacles(tiles, spawn, crystal) {
        for (let dy = -1; dy <= 1; dy++) {
            for (let dx = -1; dx <= 1; dx++) {
                const sy = spawn.y + dy;
                const sx = spawn.x + dx;
                if (sy >= 0 && sy < tiles.length && sx >= 0 && sx < tiles[0].length) {
                    tiles[sy][sx] = TileType.EMPTY;
                }
                const cy = crystal.y + dy;
                const cx = crystal.x + dx;
                if (cy >= 0 && cy < tiles.length && cx >= 0 && cx < tiles[0].length) {
                    tiles[cy][cx] = TileType.EMPTY;
                }
            }
        }
    }

    /**
     * 生成降级无障碍地图
     * @returns {GameMap}
     */
    _generateFallbackMap(w, h, seed) {
        const tiles = this._createTileGrid(w, h);
        const spawn = { x: 0, y: Math.floor(h / 2) };
        const crystal = { x: w - 1, y: Math.floor(h / 2) };

        // 直线路径
        for (let x = 0; x < w; x++) {
            tiles[spawn.y][x] = TileType.PATH;
            if (spawn.y + 1 < h) tiles[spawn.y + 1][x] = TileType.PLACEABLE;
            if (spawn.y - 1 >= 0) tiles[spawn.y - 1][x] = TileType.PLACEABLE;
        }
        tiles[spawn.y][spawn.x] = TileType.ENTRANCE;
        tiles[crystal.y][crystal.x] = TileType.CRYSTAL;

        return {
            width: w,
            height: h,
            tiles,
            path: Array.from({ length: w }, (_, x) => ({ x, y: spawn.y })),
            spawnPoint: spawn,
            crystalPoint: crystal,
            seed,
        };
    }
}
```

### 2.5 A* 寻路算法

```javascript
class AStarPathfinding {
    /**
     * 4方向A*寻路
     * @param {TileType[][]} tiles
     * @param {GridPoint} start
     * @param {GridPoint} end
     * @returns {GridPoint[]|null}
     */
    findPath(tiles, start, end) {
        /**
         * @typedef {object} Node
         * @property {GridPoint} point
         * @property {number} g
         * @property {number} h
         * @property {number} f
         * @property {Node|null} parent
         */

        const openSet = [];
        const closedSet = new Set();
        const manhattan = (a, b) => Math.abs(a.x - b.x) + Math.abs(a.y - b.y);

        openSet.push({
            point: start,
            g: 0,
            h: manhattan(start, end),
            f: manhattan(start, end),
            parent: null,
        });

        const dirs = [[0, 1], [1, 0], [0, -1], [-1, 0]];

        while (openSet.length > 0) {
            // 取 f 最小的节点
            let minIdx = 0;
            for (let i = 1; i < openSet.length; i++) {
                if (openSet[i].f < openSet[minIdx].f) minIdx = i;
            }
            const current = openSet.splice(minIdx, 1)[0];
            const key = `${current.point.x},${current.point.y}`;

            if (current.point.x === end.x && current.point.y === end.y) {
                const path = [];
                let node = current;
                while (node) {
                    path.unshift(node.point);
                    node = node.parent;
                }
                return path;
            }

            if (closedSet.has(key)) continue;
            closedSet.add(key);

            for (const [dx, dy] of dirs) {
                const nx = current.point.x + dx;
                const ny = current.point.y + dy;
                if (nx < 0 || nx >= tiles[0].length || ny < 0 || ny >= tiles.length) continue;
                if (closedSet.has(`${nx},${ny}`)) continue;
                // 障碍物不可通行（除非是终点或起点）
                if (tiles[ny][nx] === TileType.OBSTACLE) continue;

                const g = current.g + 1;
                const hVal = manhattan({ x: nx, y: ny }, end);
                openSet.push({
                    point: { x: nx, y: ny },
                    g, h: hVal, f: g + hVal,
                    parent: current,
                });
            }
        }
        return null; // 无路径
    }
}
```

## 3. 伤害计算公式

> 📌 来自需求 §5.0：伤害计算公式是数值设计的根基。

### 3.1 基础公式

```
物理伤害 = 塔攻击力 × (1 - 目标护甲 / (目标护甲 + 100))
魔法伤害 = 塔攻击力 × (1 - 目标魔抗 / (目标魔抗 + 100))
真实伤害 = 塔攻击力（不受护甲/魔抗减免）

暴击伤害 = 基础伤害 × 暴击伤害倍率（默认 1.5×）
最终伤害 = 基础伤害 × 暴击系数 × 随机浮动 (0.95 ~ 1.05)
```

### 3.2 减伤速查表

| 护甲/魔抗值 | 减伤比例 | 说明 |
|:---------:|:------:|------|
| 0 | 0% | 无减免 |
| 25 | 20% | 低护甲 |
| 50 | 33% | 中护甲（重甲怪） |
| 80 | 44% | BOSS 级护甲（岩石巨人） |
| 100 | 50% | - |
| 200 | 67% | 极端高护甲 |

### 3.3 伤害类型与减免对应

| 伤害类型 | 受护甲影响 | 受魔抗影响 | 代表塔/来源 |
|---------|:----:|:----:|--------|
| **物理伤害** | ✅ | ❌ | 箭塔、炮塔、召唤塔小兵 |
| **魔法伤害** | ❌ | ✅ | 魔法塔、电塔 |
| **真实伤害** | ❌ | ❌ | 毒塔 DOT、即死诅咒(SSR-03) |
| **混合伤害** | ✅ | ✅ | 特殊遗物效果 |

### 3.4 溅射与连锁伤害

| 机制 | 公式 | 说明 |
|------|------|------|
| **溅射伤害**（炮塔） | 主目标=100%攻击力；副目标=100%攻击力 | 主副目标伤害相同，各自独立计算护甲 |
| **连锁伤害**（电塔） | 第N跳=攻击力×(1-递减系数)^N | Lv.1 递减20%：100%→80%→64%；Lv.3 递减10%：100%→90%→81% |
| **DOT伤害**（毒塔） | 每层=DOT值/s×层数 | 真实伤害，不受减免 |

### 3.5 伤害计算实现

```javascript
/**
 * @typedef {object} DamageResult
 * @property {FixedPoint} finalDamage
 * @property {boolean} isCrit
 * @property {boolean} isInstantKill
 * @property {string} damageType - 'PHYSICAL'|'MAGIC'|'TRUE'
 */

class DamageCalculator {
    /**
     * 计算最终伤害
     * @param {FixedPoint} baseDamage     - 塔基础攻击力
     * @param {string} damageType         - 'PHYSICAL'|'MAGIC'|'TRUE'
     * @param {number} targetArmor        - 目标护甲
     * @param {number} targetMagicResist  - 目标魔抗
     * @param {number} critRate           - 暴击率 [0, 1]
     * @param {number} critDamageMul      - 暴击伤害倍率（默认1.5）
     * @param {number} instantKillRate    - 即死率 [0, 1]，来自SSR-03
     * @param {SeededRandom} rng
     * @returns {DamageResult}
     */
    static calculate(baseDamage, damageType, targetArmor, targetMagicResist,
                     critRate, critDamageMul, instantKillRate, rng) {
        critDamageMul = critDamageMul || 1.5;
        instantKillRate = instantKillRate || 0;

        // 即死判定（非BOSS）
        if (instantKillRate > 0 && rng.next() < instantKillRate) {
            return { finalDamage: FixedPoint.fromInt(999999), isCrit: false, isInstantKill: true, damageType };
        }

        // 护甲/魔抗减免
        let damage = baseDamage.clone();
        if (damageType === DamageType.PHYSICAL) {
            const reduction = targetArmor / (targetArmor + 100);
            damage = damage.multiply(FixedPoint.fromFloat(1 - reduction));
        } else if (damageType === DamageType.MAGIC) {
            const reduction = targetMagicResist / (targetMagicResist + 100);
            damage = damage.multiply(FixedPoint.fromFloat(1 - reduction));
        }
        // 真实伤害不减免

        // 暴击判定
        let isCrit = false;
        if (rng.next() < critRate) {
            damage = damage.multiply(FixedPoint.fromFloat(critDamageMul));
            isCrit = true;
        }

        // 随机浮动 (0.95 ~ 1.05)
        const floatVal = 0.95 + rng.next() * 0.10;
        damage = damage.multiply(FixedPoint.fromFloat(floatVal));

        return { finalDamage: damage, isCrit, isInstantKill: false, damageType };
    }
}

/** @enum {string} */
const DamageType = Object.freeze({
    PHYSICAL: 'PHYSICAL',
    MAGIC: 'MAGIC',
    TRUE: 'TRUE',
});
```

### 4.3 塔运行时实例

```javascript
/**
 * @enum {string} 目标优先级模式
 */
const TargetMode = Object.freeze({
    FIRST: 'FIRST',         // 最靠近水晶（终点）
    LAST: 'LAST',           // 最靠近入口
    STRONG: 'STRONG',       // HP最高
    WEAK: 'WEAK',           // HP最低
    CLOSE: 'CLOSE',         // 距离塔最近
});

class Tower {
    /**
     * @param {string} id            - 唯一ID
     * @param {string} ownerId       - 所属玩家ID
     * @param {TowerBaseConfig} config
     * @param {number} gridX
     * @param {number} gridY
     */
    constructor(id, ownerId, config, gridX, gridY) {
        this.id = id;
        this.ownerId = ownerId;
        this.type = config.type;
        this.config = config;
        this.gridX = gridX;
        this.gridY = gridY;
        this.level = 1;

        // 基础属性（动态，会累加成长和Buff）
        this.baseAttack = FixedPoint.fromFloat(config.levels[0].attack);
        this.baseAttackSpeed = config.levels[0].attackSpeed; // 秒
        this.baseRange = config.levels[0].range;

        // 运行时状态
        /** @type {Map<string, BuffInstance>} */
        this.activeBuffs = new Map();
        this.accumulatedGrowth = {};     // 累计成长记录 { attack: 15, attackSpeed: -0.25 }
        this.targetMode = TargetMode.FIRST;
        this.attackCooldown = 0;         // 当前冷却(秒)
        this.totalKills = 0;

        // 特殊属性（不同塔类型）
        this.slowPercent = 0;            // 冰塔
        this.dotDamage = 0;              // 毒塔
        this.chainCount = 0;             // 电塔
        this.splashRange = 0;            // 炮塔
        this.pierceCount = 0;            // 魔法塔
        this.auraAttackBonus = 0;        // 图腾塔
        this.minionPool = [];            // 召唤塔
    }

    // ===== 动态属性计算（含Buff叠加）=====

    /** @returns {FixedPoint} 最终攻击力 */
    get finalAttack() {
        let result = this.baseAttack.clone();
        // 乘法叠加：所有 Buff 乘算
        for (const buff of this.activeBuffs.values()) {
            for (const effect of buff.config.effects) {
                if (effect.target === 'ALL_TOWERS' && effect.stat === 'attack') {
                    if (effect.modifierType === 'MULTIPLY') {
                        result = result.multiply(FixedPoint.fromFloat(1 + effect.value));
                    } else if (effect.modifierType === 'ADD') {
                        result = result.add(FixedPoint.fromFloat(effect.value));
                    }
                }
            }
        }
        return result;
    }

    /** @returns {number} 最终攻击间隔(秒) */
    get finalAttackSpeed() {
        let speed = this.baseAttackSpeed;
        for (const buff of this.activeBuffs.values()) {
            for (const effect of buff.config.effects) {
                if (effect.stat === 'attackSpeed') {
                    if (effect.modifierType === 'MULTIPLY') {
                        speed *= (1 + effect.value);
                    }
                }
            }
        }
        return Math.max(0.2, speed); // 下限0.2秒
    }

    /** @returns {number} 最终射程(格) */
    get finalRange() {
        let range = this.baseRange;
        for (const buff of this.activeBuffs.values()) {
            for (const effect of buff.config.effects) {
                if (effect.stat === 'range') {
                    if (effect.modifierType === 'ADD') range += effect.value;
                    else if (effect.modifierType === 'MULTIPLY') range *= (1 + effect.value);
                }
            }
        }
        return Math.max(1, range);
    }

    // ===== 目标选择 =====

    /**
     * 在射程范围内按目标模式选择攻击目标
     * @param {Enemy[]} enemies - 所有存活怪物
     * @returns {{ enemy: Enemy, distance: number }|null}
     */
    findTarget(enemies) {
        /** @type {{ enemy: Enemy, distance: number }[]} */
        const inRange = [];
        for (const e of enemies) {
            if (!e.isAlive) continue;
            // 飞行怪检查
            if (e.isFlying && this.config.targetType === 'GROUND') continue;
            // 隐行怪检查
            if (e.isStealthed && !this._canDetectStealth()) continue;

            const dx = e.currentX - (this.gridX + 0.5);
            const dy = e.currentY - (this.gridY + 0.5);
            const dist = Math.sqrt(dx * dx + dy * dy);
            if (dist <= this.finalRange) {
                inRange.push({ enemy: e, distance: dist });
            }
        }

        if (inRange.length === 0) return null;

        switch (this.targetMode) {
            case TargetMode.FIRST:
                return inRange.reduce((a, b) =>
                    a.enemy.pathProgress.value > b.enemy.pathProgress.value ? a : b);
            case TargetMode.LAST:
                return inRange.reduce((a, b) =>
                    a.enemy.pathProgress.value < b.enemy.pathProgress.value ? a : b);
            case TargetMode.STRONG:
                return inRange.reduce((a, b) =>
                    a.enemy.currentHp.value > b.enemy.currentHp.value ? a : b);
            case TargetMode.WEAK:
                return inRange.reduce((a, b) =>
                    a.enemy.currentHp.value < b.enemy.currentHp.value ? a : b);
            case TargetMode.CLOSE:
                return inRange.reduce((a, b) => a.distance < b.distance ? a : b);
            default:
                return inRange[0];
        }
    }

    /**
     * 是否有反隐能力
     * @returns {boolean}
     */
    _canDetectStealth() {
        // 魔法塔 Lv.2+ 自动获得反隐
        if (this.type === TowerType.MAGIC && this.level >= 2) return true;
        // 电塔 Lv.3 解锁反隐
        if (this.type === TowerType.TESLA && this.level >= 3) return true;
        // 检查全局反隐遗物（SR遗物·真视之眼）
        if (GameWorld.instance.hasGlobalRelic('SSR-STEALTH_DETECT')) return true;
        return false;
    }

    /**
     * 主更新（每逻辑帧调用）
     * @param {number} dt - 距离上次调用的时间(秒)
     * @param {Enemy[]} enemies
     * @returns {BulletCommand[]} 本帧发射的子弹指令
     */
    update(dt, enemies) {
        this.attackCooldown -= dt;
        /** @type {BulletCommand[]} */
        const bullets = [];

        while (this.attackCooldown <= 0) {
            const target = this.findTarget(enemies);
            if (!target) break;

            this.attackCooldown += this.finalAttackSpeed;
            bullets.push(this._createBulletCommand(target.enemy));
        }

        return bullets;
    }

    /**
     * 构造子弹指令
     * @param {Enemy} target
     * @returns {BulletCommand}
     */
    _createBulletCommand(target) {
        return {
            towerId: this.id,
            towerType: this.type,
            targetId: target.id,
            damage: this.finalAttack.clone(),
            damageType: this.config.damageType,
            specialEffect: this._getSpecialEffect(),
        };
    }

    /** @returns {object|null} */
    _getSpecialEffect() {
        const lvlConfig = this.config.levels[this.level - 1];
        return lvlConfig.special || null;
    }
}
```

### 4.4 塔操作

```javascript
/** @enum {string} */
const TowerAction = Object.freeze({
    PLACE: 'PLACE',
    UPGRADE: 'UPGRADE',
    SELL: 'SELL',
    SWITCH_TARGET: 'SWITCH_TARGET',
});

class TowerController {
    constructor(battleManager) {
        /** @type {BattleManager} */
        this.battle = battleManager;
        /** @type {Map<string, Tower>} */
        this.towers = new Map();
    }

    /**
     * 建造防御塔
     * @param {string} playerId
     * @param {string} towerType
     * @param {number} gridX
     * @param {number} gridY
     * @returns {{ success: boolean, error?: string }}
     */
    placeTower(playerId, towerType, gridX, gridY) {
        const player = this.battle.getPlayer(playerId);
        const config = TOWER_CONFIG[towerType];
        if (!config) return { success: false, error: '未知塔类型' };

        // 校验金币
        const actualCost = this.battle.economy.getBuildCost(playerId, config.cost.build);
        if (player.gold < actualCost) {
            return { success: false, error: '金币不足' };
        }

        // 校验位置
        const tile = this.battle.map.tiles[gridY]?.[gridX];
        if (tile !== TileType.PLACEABLE) {
            return { success: false, error: '此位置不可放置' };
        }

        // 校验是否被占用
        if (this.battle.getTowerAt(gridX, gridY)) {
            return { success: false, error: '此位置已有塔' };
        }

        // 校验建造上限
        if (this.battle.multiplayerConfig) {
            const maxPerPlayer = this.battle.multiplayerConfig.maxTowersPerPlayer;
            if (maxPerPlayer) {
                const playerTowerCount = this._countPlayerTowers(playerId);
                if (playerTowerCount >= maxPerPlayer) {
                    return { success: false, error: `每人最多建造${maxPerPlayer}个塔` };
                }
            }
        }

        // 校验塔已解锁
        if (!player.unlockedTowers.has(towerType)) {
            return { success: false, error: '该塔尚未解锁' };
        }

        // 扣金币 + 建造
        this.battle.economy.spendGold(playerId, actualCost);
        const tower = new Tower(
            this._generateTowerId(),
            playerId,
            config,
            gridX,
            gridY,
        );
        this.towers.set(tower.id, tower);
        this.battle.map.tiles[gridY][gridX] = TileType.PATH; // 占位

        return { success: true, towerId: tower.id };
    }

    /**
     * 升级防御塔
     * @param {string} playerId
     * @param {string} towerId
     * @returns {{ success: boolean, error?: string }}
     */
    upgradeTower(playerId, towerId) {
        const tower = this.towers.get(towerId);
        if (!tower) return { success: false, error: '塔不存在' };
        if (tower.ownerId !== playerId) return { success: false, error: '不能升级队友的塔' };
        if (tower.level >= 3) return { success: false, error: '已达最高等级' };

        const upgradeCost = tower.config.cost.upgrade[tower.level - 1];
        const actualCost = this.battle.economy.getBuildCost(playerId, upgradeCost);
        if (this.battle.getPlayer(playerId).gold < actualCost) {
            return { success: false, error: '金币不足' };
        }

        this.battle.economy.spendGold(playerId, actualCost);
        tower.level++;

        // 更新基础属性到对应等级
        const lvlConfig = tower.config.levels[tower.level - 1];
        tower.baseAttack = FixedPoint.fromFloat(lvlConfig.attack + (tower.accumulatedGrowth.attack || 0));
        tower.baseAttackSpeed = lvlConfig.attackSpeed + (tower.accumulatedGrowth.attackSpeed || 0);
        tower.baseRange = lvlConfig.range + (tower.accumulatedGrowth.range || 0);

        return { success: true };
    }

    /**
     * 出售防御塔
     * @param {string} playerId
     * @param {string} towerId
     * @returns {{ success: boolean, refund: number, lostBuffs?: string[] }}
     */
    sellTower(playerId, towerId) {
        const tower = this.towers.get(towerId);
        if (!tower) return { success: false, refund: 0 };
        if (tower.ownerId !== playerId) return { success: false, refund: 0 };

        // 计算返还金币（建造费用+升级费用之和×50%，按实际消耗）
        let totalSpent = tower.config.cost.build;
        for (let i = 1; i < tower.level; i++) {
            totalSpent += tower.config.cost.upgrade[i - 1];
        }
        // 受天赋/遗物减费影响，按实际消耗
        // 简化处理：按原始价格×50%
        const refund = Math.floor(totalSpent * 0.5);

        // 收集将丢失的单体Buff
        /** @type {string[]} */
        const lostBuffs = [];
        for (const [id, buff] of tower.activeBuffs) {
            if (buff.config.scope === 'SINGLE_TOWER') {
                lostBuffs.push(buff.config.name);
            }
        }

        // 清理
        this.towers.delete(towerId);
        this.battle.map.tiles[tower.gridY][tower.gridX] = TileType.PLACEABLE; // 恢复可放置
        this.battle.economy.addGold(playerId, refund);

        return { success: true, refund, lostBuffs };
    }

    /**
     * 切换目标模式
     * @param {string} playerId
     * @param {string} towerId
     * @param {string} mode
     * @returns {boolean}
     */
    switchTargetMode(playerId, towerId, mode) {
        const tower = this.towers.get(towerId);
        if (!tower || tower.ownerId !== playerId) return false;
        if (!Object.values(TargetMode).includes(mode)) return false;

        // 需天赋解锁（默认仅 FIRST）
        if (!this.battle.getPlayer(playerId).hasTalent('TACTICAL_MASTER')) {
            if (mode !== TargetMode.FIRST) return false;
        }

        tower.targetMode = mode;
        return true;
    }

    /** @param {string} playerId @returns {Tower[]} */
    getPlayerTowers(playerId) {
        const result = [];
        for (const t of this.towers.values()) {
            if (t.ownerId === playerId) result.push(t);
        }
        return result;
    }

    /** @param {string} playerId @returns {number} */
    _countPlayerTowers(playerId) {
        let count = 0;
        for (const t of this.towers.values()) {
            if (t.ownerId === playerId) count++;
        }
        return count;
    }

    _generateTowerId() {
        return 't_' + Date.now().toString(36) + '_' + Math.random().toString(36).slice(2, 6);
    }
}
```

```javascript
class Enemy {

     * @param {string} id
     * @param {EnemyBaseConfig} config
     * @param {number} baseHp           - 基准HP（难度调整后）
     * @param {number} baseSpeed        - 基准移速

        this.id = id;
        this.type = config.type;

        // HP

        this.currentHp = this.maxHp.clone();
        this.armor = config.armor;

        this.reward = config.reward;
        this.crystalDamage = config.crystalDamage;
        this.isFlying = config.movementType === 'AIR';
        this.isStealthed = config.type === EnemyType.STEALTH;

        // 路径
        this.path = path;
        this.pathProgress = FixedPoint.ZERO; // 当前段内的进度 [0, 1)
        this._speed = FixedPoint.fromFloat(baseSpeed * config.speedMul);
        this.currentX = path[0].x + 0.5;
        this.currentY = path[0].y + 0.5;

        // 状态
        this.isAlive = true;
        this.isSlowed = false;
        this.slowPercent = 0;
        this.slowTimer = 0;
        this.isStunned = false;
        this.stunTimer = 0;

        // DOT
        this.dotStacks = 0;
        this.dotDamagePerSec = 0;
        this.dotTimer = 0;

        // 护盾
        this.shieldHp = config.type === EnemyType.SHIELD
            ? FixedPoint.fromFloat(baseHp * config.hpMul) // 护盾=HP
            : FixedPoint.ZERO;

        // 特殊行为计时
        this._specialTimer = 0;
    }

    /**
     * 受到伤害
     * @param {FixedPoint} rawDamage
     * @param {string} damageType
     * @param {boolean} isCrit
     * @param {boolean} isInstantKill
     * @returns {{ actualDamage: FixedPoint, killed: boolean, shieldBroken: boolean }}
     */
    takeDamage(rawDamage, damageType, isCrit, isInstantKill) {
        if (!this.isAlive) return { actualDamage: FixedPoint.ZERO, killed: false, shieldBroken: false };

        let damage = rawDamage.clone();
        let shieldBroken = false;

        // 即死
        if (isInstantKill && this.type !== EnemyType.BOSS) {
            this.currentHp = FixedPoint.ZERO;
            this.isAlive = false;
            return { actualDamage: this.maxHp.clone(), killed: true, shieldBroken };
        }

        // 护盾吸收
        if (this.shieldHp.value > 0) {
            if (this.shieldHp.greaterOrEqual(damage)) {
                this.shieldHp = this.shieldHp.subtract(damage);
                return { actualDamage: damage, killed: false, shieldBroken: false };
            }
            damage = damage.subtract(this.shieldHp);
            this.shieldHp = FixedPoint.ZERO;
            shieldBroken = true;
        }

        // 护甲/魔抗减免
        let actualDamage = damage;
        if (damageType !== DamageType.TRUE) {
            const resist = damageType === DamageType.PHYSICAL ? this.armor : this.magicResist;
            const reduction = resist / (resist + 100);
            actualDamage = damage.multiply(FixedPoint.fromFloat(1 - reduction));
        }

        this.currentHp = this.currentHp.subtract(actualDamage);
        const killed = this.currentHp.value <= 0;
        if (killed) {
            this.currentHp = FixedPoint.ZERO;
            this.isAlive = false;
        }

        return { actualDamage, killed, shieldBroken };
    }

    /**
     * 主更新
     * @param {number} dt - 秒
     * @returns {{ reachedEnd: boolean, spawnedChildren: Enemy[] }}
     */
    update(dt) {
        if (!this.isAlive) return { reachedEnd: false, spawnedChildren: [] };

        // 眩晕
        if (this.isStunned) {
            this.stunTimer -= dt;
            if (this.stunTimer <= 0) this.isStunned = false;
            return { reachedEnd: false, spawnedChildren: [] };
        }

        // DOT
        if (this.dotStacks > 0 && this.dotDamagePerSec > 0) {
            this.dotTimer += dt;
            while (this.dotTimer >= 1.0) {
                this.dotTimer -= 1.0;
                const dotDamage = FixedPoint.fromFloat(this.dotDamagePerSec * this.dotStacks);
                const { killed } = this.takeDamage(dotDamage, DamageType.TRUE, false, false);
                if (killed) return { reachedEnd: false, spawnedChildren: [] };
            }
        }

        // 减速计时
        if (this.isSlowed) {
            this.slowTimer -= dt;
            if (this.slowTimer <= 0) {
                this.isSlowed = false;
                this.slowPercent = 0;
            }
        }

        // 再生（M-06怪物增益）
        if (this._hasRegen) {
            this.currentHp = this.currentHp.add(
                this.maxHp.multiply(FixedPoint.fromFloat(0.01 * dt))
            );
            if (this.currentHp.value > this.maxHp.value) {
                this.currentHp = this.maxHp.clone();
            }
        }

        // 移动
        const effectiveSpeed = this.isSlowed
            ? this._speed.multiply(FixedPoint.fromFloat(1 - this.slowPercent))
            : this._speed;

        const moveDistance = effectiveSpeed.multiply(FixedPoint.fromFloat(dt));
        this.pathProgress = this.pathProgress.add(moveDistance);

        // 段切换
        while (this.pathProgress.greaterOrEqual(FixedPoint.ONE)) {
            this.pathProgress = this.pathProgress.subtract(FixedPoint.ONE);
            const nextIdx = this._currentPathIndex() + 1;
            if (nextIdx >= this.path.length) {
                // 到达终点
                this.isAlive = false;
                return { reachedEnd: true, spawnedChildren: [] };
            }
        }

        // 插值坐标
        const idx = this._currentPathIndex();
        const curr = this.path[idx];
        const next = this.path[Math.min(idx + 1, this.path.length - 1)];
        const t = this.pathProgress.toFloat();
        this.currentX = curr.x + (next.x - curr.x) * t + 0.5;
        this.currentY = curr.y + (next.y - curr.y) * t + 0.5;

        return { reachedEnd: false, spawnedChildren: [] };
    }

    _currentPathIndex() {
        // 计算当前在路径上的索引
        let progress = this.pathProgress.value;
        let idx = 0;
        // 简化：直接从pathProgress推断
        // 实际实现中需要跟踪当前段索引
        return idx;
    }

    /** @param {number} percent @param {number} durationSec */
    applySlow(percent, durationSec) {
        this.isSlowed = true;
        this.slowPercent = Math.max(this.slowPercent, percent);
        this.slowTimer = Math.max(this.slowTimer, durationSec);
    }

    /** @param {number} dps @param {number} maxStacks */
    applyDot(dps, maxStacks) {
        if (this.dotStacks < maxStacks) this.dotStacks++;
        this.dotDamagePerSec = Math.max(this.dotDamagePerSec, dps);
    }

    /** @param {number} durationSec */
    applyStun(durationSec) {
        this.isStunned = true;
        this.stunTimer = Math.max(this.stunTimer, durationSec);
    }

    /** 启用再生（来自怪物增益 M-06） */
    enableRegen() { this._hasRegen = true; }
}
```

### 6.5 毒塔 DOT 叠加模型 `🆕`

> 📌 来自需求 §5.1.6 补充：毒塔 DOT 叠加的精确数学模型。

```javascript
/**
 * DOT 数据模型
 * 每个 DOT 来源独立计时，不同层可以有不同剩余时间
 */
class DotManager {
    /**
     * @param {number} maxStacks - 最大叠加层数
     */
    constructor(maxStacks) {
        this.maxStacks = maxStacks;
        /** @type {{ towerId: string, damagePerSec: FixedPoint, remainingTime: number, appliedAt: number }[]} */
        this.stacks = [];
    }

    /**
     * 叠加一层 DOT
     * @param {string} towerId      - 来源塔ID
     * @param {FixedPoint} damagePerSec - 每秒伤害
     * @param {number} duration     - 持续秒数
     * @returns {boolean} 是否成功叠加
     */
    addStack(towerId, damagePerSec, duration) {
        // 已达上限
        if (this.stacks.length >= this.maxStacks) {
            // 替换最旧的层
            this.stacks.shift();
        }

        this.stacks.push({
            towerId,
            damagePerSec: damagePerSec.clone(),
            remainingTime: duration,
            appliedAt: Date.now(),
        });
        return true;
    }

    /**
     * 每帧更新 DOT 计时 + 计算伤害
     * @param {number} dt - 秒
     * @returns {FixedPoint} 本帧总 DOT 伤害
     */
    update(dt) {
        let totalDamage = FixedPoint.ZERO;

        this.stacks = this.stacks.filter(stack => {
            // 每层独立计时
            const frameDamage = stack.damagePerSec.multiply(FixedPoint.fromFloat(dt));
            totalDamage = totalDamage.add(frameDamage);

            stack.remainingTime -= dt;
            return stack.remainingTime > 0; // 到期自动移除
        });

        return totalDamage;
    }

    /**
     * 毒塔 Lv.3 毒爆：清除所有 DOT 并结算剩余伤害
     * @returns {FixedPoint}
     */
    detonate() {
        let remainingDamage = FixedPoint.ZERO;
        for (const stack of this.stacks) {
            const remaining = stack.damagePerSec.multiply(
                FixedPoint.fromFloat(stack.remainingTime)
            );
            remainingDamage = remainingDamage.add(remaining);
        }
        this.stacks = [];
        return remainingDamage;
    }

    /** @returns {number} 当前层数 */
    get currentStacks() { return this.stacks.length; }

    /** @returns {FixedPoint} 当前每秒总伤害 */
    get totalDps() {
        let total = FixedPoint.ZERO;
        for (const stack of this.stacks) {
            total = total.add(stack.damagePerSec);
        }
        return total;
    }
}
```

```javascript
class CrystalController {
    /**
     * @param {number} initialHp
     * @param {number} maxHp
     */
    constructor(initialHp, maxHp) {
        this.currentHp = initialHp;
        this.maxHp = maxHp;

        return this.currentHp <= 0;

    addShield(layers) {
        this.shieldLayers += layers;
    }

    /** @returns {boolean} */
    get isDanger() { return this.currentHp <= 2; }

    /** @returns {boolean} */
    get isDestroyed() { return this.currentHp <= 0; }
```

### 8.3 怪物路径热力图

> 建造塔时（拖拽预览状态），地图上显示怪物经过频率的热力图。

```javascript
class PathHeatmap {
    /**
     * 生成热力图数据
     * 统计路径上每个格子的通过频率
     * @param {GridPoint[]} path
     * @returns {Map<string, number>} "x,y" → 频率值
     */
    static generate(path) {
        /** @type {Map<string, number>} */
        const heatmap = new Map();
        
        // 路径主干频率最高
        for (const p of path) {
            const key = `${p.x},${p.y}`;
            heatmap.set(key, (heatmap.get(key) || 0) + 1);
        }
        
        // 路径邻近格也有较低热度（溅射范围）
        for (const p of path) {
            for (let dy = -1; dy <= 1; dy++) {
                for (let dx = -1; dx <= 1; dx++) {
                    const key = `${p.x + dx},${p.y + dy}`;
                    heatmap.set(key, (heatmap.get(key) || 0) + 0.5);
                }
            }
        }
        
        return heatmap;
    }
    
    /**
     * 渲染热力图
     * @param {Map<string, number>} heatmap
     */
    static render(heatmap) {
        const maxFreq = Math.max(...heatmap.values());
        for (const [key, freq] of heatmap) {
            const [x, y] = key.split(',').map(Number);
            const normalizedFreq = freq / maxFreq;
            
            // 颜色渐变：蓝色（低）→ 绿色（中）→ 红色（高）
            const color = this._lerpColor(normalizedFreq);
            const alpha = 0.3 + normalizedFreq * 0.4; // 透明度 0.3~0.7
            
            this._drawTile(x, y, color, alpha);
        }
    }
    
    /**
     * 频率值→颜色
     * @param {number} t - [0, 1]
     * @returns {string} hex color
     */
    static _lerpColor(t) {
        if (t < 0.5) {
            // 蓝 → 绿
            const r = 0;
            const g = Math.round(t * 2 * 255);
            const b = Math.round((1 - t * 2) * 255);
            return `#${r.toString(16).padStart(2,'0')}${g.toString(16).padStart(2,'0')}${b.toString(16).padStart(2,'0')}`;
        } else {
            // 绿 → 红
            const r = Math.round((t - 0.5) * 2 * 255);
            const g = Math.round((1 - t) * 2 * 255);
            const b = 0;
            return `#${r.toString(16).padStart(2,'0')}${g.toString(16).padStart(2,'0')}${b.toString(16).padStart(2,'0')}`;
        }
    }
}
```

**热力图显示规则**：
- 仅在塔拖拽预览时显示，松手后消失
- 帮助玩家快速识别「咽喉要道」
- 高频率路径交叉点是最佳建塔位置
