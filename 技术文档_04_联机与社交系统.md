```javascript
/**
 * 微信授权登录管理器
 * 依赖 wx.login、wx.getUserProfile（新版API）
 */
class AuthManager {
    constructor() {
        /** @type {UserProfile|null} */
        this.currentUser = null;
        this.token = null;
    }

    /**
     * 执行微信登录
     * @returns {Promise<UserProfile>}
     */
    async login() {
        // Step 1: wx.login 获取临时code
        const loginRes = await this._wxLogin();

        // Step 2: 服务端用code换取openid+session_key
        const serverRes = await this._serverAuth(loginRes.code);

        if (!serverRes.success) {
            // 降级为游客模式
            return this._loginAsGuest();
        }

        // Step 3: 获取用户信息（头像昵称）
        const userInfo = await this._getUserProfile();

        // Step 4: 构建用户档案
        this.currentUser = {
            openid: serverRes.openid,
            nickName: userInfo.nickName,
            avatarUrl: userInfo.avatarUrl,
            level: serverRes.level || 1,
            exp: serverRes.exp || 0,
            gold: serverRes.gold || 0,
            diamond: serverRes.diamond || 0,
            isGuest: false,
        };
        this.token = serverRes.token;

        return this.currentUser;
    }

    /**
     * @returns {Promise<{code: string}>}
     */
    _wxLogin() {
        return new Promise((resolve, reject) => {
            wx.login({
                success: resolve,
                fail: reject,
            });
        });
    }

    /**
     * @returns {Promise<UserProfile>}
     */
    _getUserProfile() {
        return new Promise((resolve, reject) => {
            wx.getUserProfile({
                desc: '用于游戏内展示头像和昵称',
                success: (res) => resolve(res.userInfo),
                fail: () => resolve({ nickName: '游客', avatarUrl: '' }),
            });
        });
    }

    /**
     * 服务端认证
     * @param {string} code
     * @returns {Promise<object>}
     */
    async _serverAuth(code) {
        const response = await fetch('https://api.xxx.com/auth/login', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ code }),
        });
        return response.json();
    }

    /**
     * 游客模式
     * @returns {UserProfile}
     */
    _loginAsGuest() {
        this.currentUser = {
            openid: 'guest_' + Date.now(),
            nickName: '游客' + Math.random().toString(36).slice(2, 6),
            avatarUrl: '',
            level: 1, exp: 0, gold: 0, diamond: 0,
            isGuest: true,
        };
        return this.currentUser;
    }
}
```

```javascript
class FriendManager {
    constructor() {
        /** @type {FriendInfo[]} */
        this.friends = [];
        this.isLoaded = false;
    }

    /**
     * 加载好友列表（合并微信关系链+服务端数据）
     * @returns {Promise<FriendInfo[]>}
     */
    async loadFriends() {
        // 1. 获取微信好友中玩本游戏的人
        const wxFriendRes = await this._getWxFriendList();

        // 2. 服务端匹配游戏内玩家数据
        const serverRes = await this._queryGameFriends(wxFriendRes.openids);

        // 3. 合并排序（在线 > 对局中 > 离线）
        this.friends = serverRes.friends
            .map(f => ({
                openid: f.openid,
                nickName: f.nickName,
                avatarUrl: f.avatarUrl,
                status: f.status,        // 'online'|'playing'|'offline'
                level: f.level,
                rank: f.rank,
                lastOnlineTime: f.lastOnlineTime,
                canSpectate: f.status === 'playing' && f.allowSpectate,
            }))
            .sort((a, b) => {
                const order = { online: 0, playing: 1, offline: 2 };
                return order[a.status] - order[b.status];
            });

        this.isLoaded = true;
        return this.friends;
    }

    /** @returns {Promise<{openids: string[]}>} */
    async _getWxFriendList() {
        // 微信关系链API
        return new Promise((resolve) => {
            wx.getFriendCloudStorage({
                keyList: ['openid'],
                success: resolve,
                fail: () => resolve({ data: [] }),
            });
        });
    }

    /** @param {string} friendOpenid @returns {boolean} */
    canInvite(friendOpenid) {
        const friend = this.friends.find(f => f.openid === friendOpenid);
```

```javascript
class RoomManager {
    constructor() {
        /** @type {RoomState|null} */
        this.currentRoom = null;
    }

    /**
     * 创建房间
     * @param {CreateRoomOptions} options
     * @returns {Promise<RoomState>}
     */
    async createRoom(options) {
        const msg = createMessage('create_room', {
            difficulty: options.difficulty,
            maxPlayers: options.maxPlayers,
            mapSeed: options.mapSeed || 0,
            enableVoice: options.enableVoice || false,
        });
        const response = await this._sendAndWait(msg, 'create_room_rsp');
        this.currentRoom = response.data;
        return this.currentRoom;
    }

    /**
     * 加入房间
     * @param {string} roomId - 6位房间号
     * @returns {Promise<RoomState>}
     */
    async joinRoom(roomId) {
        const msg = createMessage('join_room', { roomId });
        const response = await this._sendAndWait(msg, 'join_room_rsp');

        if (!response.success) {
            throw new Error(response.error);
        }
        this.currentRoom = response.data;
        return this.currentRoom;
    }

    /**
     * 退出房间
     */
    async leaveRoom() {
        if (!this.currentRoom) return;
        await this._send({ type: 'leave_room', roomId: this.currentRoom.roomId });
        this.currentRoom = null;
    }

    /**
     * 踢人（仅房主）
     * @param {string} targetPlayerId
     */
    async kickPlayer(targetPlayerId) {
        if (!this.currentRoom || !this._isHost()) return;
        await this._send({
            type: 'kick_player',
            roomId: this.currentRoom.roomId,
            data: { targetPlayerId },
        });
    }

    /**
     * 切换准备状态
     */
    async toggleReady() {
        if (!this.currentRoom) return;
        await this._send({
            type: 'player_ready',
            roomId: this.currentRoom.roomId,
        });
    }

    /**
     * 处理 room_update 推送
     * @param {object} data
     */
    onRoomUpdate(data) {
        if (!this.currentRoom) return;
        Object.assign(this.currentRoom, data);
        EventBus.emit('room:updated', this.currentRoom);
    }

    /** @returns {boolean} */
    _isHost() {
        return this.currentRoom &&
            this.currentRoom.players.find(p => p.isHost)?.id === AuthManager.instance.currentUser.openid;
    }

    async _send(msg) {
        WebSocketClient.instance.send(msg);
    }

    async _sendAndWait(msg, expectedType) {
        return new Promise((resolve) => {
            const handler = (response) => {
                if (response.type === expectedType) {
                    EventBus.off(expectedType, handler);
                    resolve(response);
                }
            };
            EventBus.on(expectedType, handler);
            WebSocketClient.instance.send(msg);
        });
    }
}
```

### 1.4 快速匹配

```javascript
class MatchmakingManager {
    constructor() {
        /** @type {'idle'|'matching'|'matched'|'cancelled'} */
        this.status = 'idle';
        /** @type {number} 已等待秒数 */
        this.elapsedSeconds = 0;
        this._timer = null;
    }

    /**
     * 开始匹配
     * @param {number} preferredDifficulty - 1|2|3|0(任意)
     * @returns {Promise<MatchResult>}
     */
    async startMatch(preferredDifficulty) {
        this.status = 'matching';
        this.elapsedSeconds = 0;

        this._timer = setInterval(() => {
            this.elapsedSeconds++;
            EventBus.emit('match:elapsed', this.elapsedSeconds);
        }, 1000);

        return new Promise((resolve) => {
            // 监听匹配成功
            const successHandler = (data) => {
                this._cleanup();
                this.status = 'matched';
                resolve({ success: true, room: data.room });
            };
            EventBus.on('match_rsp', successHandler);

            // 发送匹配请求
            WebSocketClient.instance.send(createMessage('start_match', {
                preferredDifficulty,
            }));

            // 超时处理
            setTimeout(() => {
                if (this.status === 'matching') {
                    // 15秒超时→提示AI队友
                    EventBus.off('match_rsp', successHandler);
                    this._cleanup();
                    this.status = 'idle';
                    resolve({ success: false, reason: 'timeout', offerAI: true });
                }
            }, 15000);
        });
    }

    /**
     * 接受AI队友
     */
    async acceptAITeammate() {
        WebSocketClient.instance.send(createMessage('accept_ai_fill', {}));
    }

    /** 取消匹配 */
    cancelMatch() {
        this._cleanup();
        this.status = 'cancelled';
        WebSocketClient.instance.send(createMessage('cancel_match', {}));
    }

    _cleanup() {
        if (this._timer) {
            clearInterval(this._timer);
            this._timer = null;
        }
    }
}

/**
 * @typedef {object} MatchResult
 * @property {boolean} success
 * @property {RoomState} [room]
 * @property {string} [reason]
 * @property {boolean} [offerAI]
 */
```

## 2. 暂停系统（含联机投票）

```javascript
class PauseController {
    constructor() {
        /** @type {'running'|'paused'|'voting'} */
        this.status = 'running';
        this.pauseCount = 0;           // 本局已暂停次数
        this.maxPauses = 3;            // 每局上限
        this.pauseTimer = null;
        /** @type {Map<string, 'yes'|'no'|null>} */
        this.votes = new Map();
    }

    /**
     * 房主暂停
     */
    hostPause() {
        if (this.status !== 'running') return;
        if (this.pauseCount >= this.maxPauses) {
            ToastUI.show('本局暂停次数已用完');
            return;
        }

        this._doPause('host');
    }

    /**
     * 队员请求暂停（发起投票）
     */
    requestPause() {
        if (this.status !== 'running') return;
        if (this.pauseCount >= this.maxPauses) {
            ToastUI.show('本局暂停次数已用完');
            return;
        }

        this.status = 'voting';
        this.votes.clear();

        WebSocketClient.instance.send(createMessage('pause_vote', {
            action: 'request',
        }));

        PauseVoteUI.show();
    }

    /**
     * 投票
     * @param {string} playerId
     * @param {'yes'|'no'} vote
     */
    onVote(playerId, vote) {
        this.votes.set(playerId, vote);
    }

    /**
     * 检查投票结果
     * @param {number} totalPlayers
     * @returns {boolean} 是否通过
     */
    checkVoteResult(totalPlayers) {
        let yesCount = 0;
        let noCount = 0;
        for (const [id, vote] of this.votes) {
            if (vote === 'yes') yesCount++;
            else if (vote === 'no') noCount++;
        }

        // 房主1票算2票
        const hostId = RoomManager.instance.currentRoom?.players.find(p => p.isHost)?.id;
        if (hostId && this.votes.get(hostId) === 'yes') yesCount++;
        else if (hostId && this.votes.get(hostId) === 'no') noCount++;

        // 超过50%同意即暂停
        if (yesCount > totalPlayers / 2) {
            this._doPause('vote');
            return true;
        }
        // 超过50%反对或全投
        if (noCount > totalPlayers / 2 || this.votes.size >= totalPlayers) {
            this.status = 'running';
            return false;
        }
        return false; // 等待
    }

    /**
     * 执行暂停
     */
    _doPause(reason) {
        this.status = 'paused';
        this.pauseCount++;
        BattleManager.instance.pause();

        // 60秒超时自动继续
        this.pauseTimer = setTimeout(() => {
            this.resume();
        }, 60000);

        EventBus.emit('pause:activated', { reason, remainingPauses: this.maxPauses - this.pauseCount });
    }

    /**
     * 恢复
     */
    resume() {
        if (this.status !== 'paused') return;
        if (this.pauseTimer) {
            clearTimeout(this.pauseTimer);
            this.pauseTimer = null;
        }
        this.status = 'running';
        BattleManager.instance.resume();
        EventBus.emit('pause:resumed');
    }

    /** @returns {boolean} */
    get canPause() {
        return this.status === 'running' &&
            this.pauseCount < this.maxPauses &&
            !WaveController.instance.isBossWave;
    }
}
```

```javascript
class ReconnectHandler {
    /**
     * 处理断线
     * @param {string} playerId
     */
    onPlayerDisconnect(playerId) {
        const player = GameWorld.instance.getPlayer(playerId);
        if (!player) return;

        // 显示断线标记
        BattleHUD.instance.showPlayerDisconnected(playerId);

        // 启动10秒倒计时
        let timer = 10;
        const interval = setInterval(() => {
            timer--;
            if (timer <= 0) {
                clearInterval(interval);
                // 超时→AI接管
                this._activateAI(playerId);
            }
        }, 1000);

        // 保存清理引用
        player._disconnectInterval = interval;
    }

    /**
     * 处理重连
     * @param {string} playerId
     * @param {object} reconnectData
     */
    async onPlayerReconnect(playerId, reconnectData) {
        const player = GameWorld.instance.getPlayer(playerId);
        if (!player) return false;

        // 清除超时定时器
        if (player._disconnectInterval) {
            clearInterval(player._disconnectInterval);
        }

        // 如果AI已激活，收回控制权
        if (player._aiActive) {
            this._deactivateAI(playerId);
        }

        // 追帧
        await FrameSyncManager.instance.catchUp(reconnectData.frameHistory);
        BattleHUD.instance.showPlayerReconnected(playerId);
        return true;
    }

    /**
     * 激活AI托管
     * @param {string} playerId
     */
    _activateAI(playerId) {
        const player = GameWorld.instance.getPlayer(playerId);
        player._aiActive = true;
        player._aiController = new AutoTowerAI(player);

        BattleHUD.instance.showPlayerAIActive(playerId);
        EventBus.emit('player:ai_activated', playerId);
    }

    /**
     * 解除AI托管
     * @param {string} playerId
     */
    _deactivateAI(playerId) {
        const player = GameWorld.instance.getPlayer(playerId);
        player._aiActive = false;
        player._aiController = null;
    }
}

/**
 * AI托管逻辑
 * 自动升级最低等级塔，不建新塔，不使用道具
 */
class AutoTowerAI {
    /** @param {Player} player */
    constructor(player) {
        this.player = player;
    }

    /**
     * AI决策（每逻辑帧检查）
     * @param {TowerController} towerCtrl
     */
    update(towerCtrl) {
        const towers = towerCtrl.getPlayerTowers(this.player.id);
        if (towers.length === 0) return;

        // 自动升级最低等级的塔（如果金币足够）
        const upgradable = towers.filter(t =>
            t.level < 3 &&
            this.player.gold >= t.config.cost.upgrade[t.level - 1]
        );
        if (upgradable.length > 0) {
            const lowestLevel = upgradable.reduce((a, b) => a.level < b.level ? a : b);
            towerCtrl.upgradeTower(this.player.id, lowestLevel.id);
        }
    }
}
```

```javascript
    addGold(playerId, amount) {
        this.goldMap.set(playerId, this.getGold(playerId) + amount);
        EventBus.emit('economy:gold_changed', { playerId, amount, newTotal: this.getGold(playerId) });

    /** @param {string} playerId @param {number} amount @returns {boolean} */
    spendGold(playerId, amount) {
        const current = this.getGold(playerId);
        if (current < amount) return false;
        this.goldMap.set(playerId, current - amount);
        EventBus.emit('economy:gold_changed', { playerId, amount: -amount, newTotal: this.getGold(playerId) });
        return true;
    }

    /**
     * 获取建造费用（含天赋/遗物减费）
     * @param {string} playerId
     * @param {number} baseCost
     * @returns {number}

        // 天赋：建造费用-5%/10%/15%
        const player = GameWorld.instance.getPlayer(playerId);
        if (player.hasTalent('BUILD_COST_1')) cost *= 0.95;
        if (player.hasTalent('BUILD_COST_2')) cost *= 0.90;
        if (player.hasTalent('BUILD_COST_3')) cost *= 0.85;

        // 遗物：SR-04 建造费-20%
        if (RelicSystem.instance.hasRelic(playerId, 'SR-04')) cost *= 0.80;

        // 骰子Buff：B-04 建造费-30%
        // ... 从 BuffManager 获取

        return Math.round(cost);
    }

    /**
     * 金币赠送
     * @param {string} fromPlayerId
     * @param {string} toPlayerId
     * @param {number} amount
     * @returns {{ success: boolean, error?: string }}
     */
    giftGold(fromPlayerId, toPlayerId, amount) {
        // 防护规则（来自需求 §11.6）
        if (fromPlayerId === toPlayerId) return { success: false, error: '不能赠送给自己' };

        const fromGold = this.getGold(fromPlayerId);

        // 单次不超过持有50% → 二次确认
        if (amount > fromGold * 0.5) {
            // UI层会弹出二次确认，此处仅记录
        }

        if (!this.spendGold(fromPlayerId, amount)) {
            return { success: false, error: '金币不足' };
        }

        // 80%到账
        const received = Math.floor(amount * 0.8);
        this.addGold(toPlayerId, received);

        // 赠送后5秒内退出→赠送失效（由服务端保护）

        return { success: true };

     * @param {string} playerId
     * @param {number} waveNumber
     * @param {number} difficulty

    waveEndReward(playerId, waveNumber, difficulty) {

        let reward = baseReward;

        // 难度影响
        const diffMod = { 1: 1.2, 2: 1.0, 3: 0.9 }[difficulty] || 1.0;
        reward *= diffMod;

        // Buff影响
        const player = GameWorld.instance.getPlayer(playerId);
        const buffs = BuffManager.instance.getPlayerBuffs(playerId);
        for (const buff of buffs) {
            for (const effect of buff.config.effects) {

                }
            }
        }

        // 遗物：SR-02 每波+150
        if (RelicSystem.instance.hasRelic(playerId, 'SR-02')) {
            reward += 150;
        }

        reward = Math.round(reward);
        this.addGold(playerId, reward);
        return reward;
    }
```

```javascript
/**
 * 对局结算评分
 * @param {GameResult} gameResult
 * @returns {{ score: number, grade: string, details: object }}
 */
function calculateScore(gameResult) {
    const weights = {
        wavesSurvived: 0.50,    // 存活波次
        remainingHp: 0.20,      // 剩余生命
        killEfficiency: 0.15,   // 击杀效率
        economyEfficiency: 0.10,// 经济效率
        specialBonus: 0.05,     // 特殊加分
    };

    const wavesScore = (gameResult.wavesSurvived / 20) * 100;
    const hpScore = (gameResult.crystalFinalHp / gameResult.crystalInitialHp) * 100;
    const killScore = (gameResult.totalKills / gameResult.totalEnemies) * 100;
    const economyScore = (gameResult.remainingGold / gameResult.totalGoldEarned) * 100;

    // 特殊加分
    let specialScore = 0;
    if (gameResult.bossNoDamage) specialScore += 2;
    if (gameResult.noLeak) specialScore += 1;
    if (!gameResult.teammateDebuffHit) specialScore += 1;
    if (gameResult.noDisconnect) specialScore += 1;
    specialScore = (specialScore / 5) * 100;

    const totalScore =
        wavesScore * weights.wavesSurvived +
        hpScore * weights.remainingHp +
        killScore * weights.killEfficiency +
        economyScore * weights.economyEfficiency +
        specialScore * weights.specialBonus;

    // 评分等级
    let grade;
    if (totalScore >= 95) grade = 'S';
    else if (totalScore >= 85) grade = 'A';
    else if (totalScore >= 70) grade = 'B';
    else if (totalScore >= 50) grade = 'C';
    else grade = 'D';

    return {
        score: Math.round(totalScore),
        grade,
        details: { wavesScore, hpScore, killScore, economyScore, specialScore },
    };
}
```

```javascript
     * @returns {boolean}

    unlock(talentId) {
        const talent = this._findTalent(talentId);
        if (!talent) return false;

        const currentLevel = this.unlocked.get(talentId) || 0;
        if (currentLevel >= talent.maxLevel) return false;

        const cost = Array.isArray(talent.cost) ? talent.cost[currentLevel] : talent.cost;
        if (this.availablePoints < cost) return false;

        // 检查前置条件
        for (const req of talent.requires) {
            if (!this.unlocked.has(req)) return false;
        }

        this.availablePoints -= cost;
        this.unlocked.set(talentId, currentLevel + 1);
        return true;
    }

    /**
     * 重置天赋树（首次免费，之后消耗钻石）

            return { success: false, cost };

        // 退还所有天赋点
        this.availablePoints = currentLevel; // 每级1点
        UserDataManager.instance.set('talentResetCount', resetCount + 1);

        if (cost > 0) {
            UserDataManager.instance.deductDiamond(cost);
        }
        return { success: true };
```

```javascript
class AchievementDataManager {

     * 更新成就进度
     * @param {string} achievementId
     * @param {number} delta

    updateProgress(achievementId, delta) {
        const progress = this._getOrCreate(achievementId);
        const config = ACHIEVEMENT_CONFIG[achievementId];

        const oldValue = progress.current;
        progress.current = Math.min(progress.current + delta, config.target);
```

```javascript
                    target.removeEventListener('click', handler);
                    this._nextStep();
                };
                target.addEventListener('click', handler);
            }
        }

        if (this.currentStep >= this.steps.length) {
            this._complete();
        } else {
            this._showStep(this.currentStep);
        }
```

```javascript
class BanNotificationHandler {
    constructor() {
        this._listenForBanEvents();

    _listenForBanEvents() {

     * 展示封禁弹窗
     * @param {object} banInfo
     * @param {string} banInfo.reason        - 封禁原因（外挂/恶意挂机/刷金币/骚扰/辱骂/其他）

     * @param {string} [banInfo.remark]      - 管理员备注

                `类型：${endText}`,
```

```javascript
/**
 * 活动弹窗管理器
 * 从服务端拉取活动配置，按规则展示弹窗
 */
class ActivityPopupManager {
    constructor() {
        /** @type {ActivityConfig[]} */
        this.activeActivities = [];
        /** @type {Set<string>} 已展示过的活动ID（本次会话） */
        this.shownToday = new Set();
    }

    /**
     * 大厅加载完成后调取活动列表
     */
    async loadActivities() {
        const response = await fetch('/api/activity/active');
        const { activities } = await response.json();
        this.activeActivities = activities;

        // 按优先级排序
        this.activeActivities.sort((a, b) => a.priority - b.priority);

        // 逐一弹出（如有多个活动）
        for (const activity of this.activeActivities) {
            if (this._shouldShow(activity)) {
                await this._showActivityPopup(activity);
                // 用户关闭后继续下一个
            }
        }
    }

    /**
     * 判断是否展示
     * @param {ActivityConfig} activity
     * @returns {boolean}
     */
    _shouldShow(activity) {
        // 1. 时间范围
        const now = Date.now();
        if (now < activity.startTime || now > activity.endTime) return false;

        // 2. 每日频控（同一活动每日最多展示1次）
        const lastShown = wx.getStorageSync(`activity_shown_${activity.id}`);
        if (lastShown && this._isSameDay(lastShown, now)) return false;

        // 3. 目标用户检查
        if (activity.targetTags) {
            const userTags = UserDataManager.instance.getTags();
            const matched = activity.targetTags.some(tag => userTags.includes(tag));
            if (!matched) return false;
        }

        return true;
    }

    /**
     * 展示活动弹窗
     * @param {ActivityConfig} activity
     * @returns {Promise<void>}
     */
    _showActivityPopup(activity) {
        return new Promise((resolve) => {
            wx.showModal({
                title: activity.popupTitle,
                content: activity.popupContent,       // 支持简单 Markdown
                confirmText: activity.buttonText || '前往查看',
                cancelText: activity.cancelText || '关闭',
                success: (res) => {
                    // 记录展示
                    wx.setStorageSync(`activity_shown_${activity.id}`, Date.now());
                    Analytics.track('activity_popup_show', { activityId: activity.id });

                    if (res.confirm) {
                        // 跳转到活动页面 / 指定功能入口
                        Analytics.track('activity_popup_click', { activityId: activity.id });
                        if (activity.jumpUrl) {
                            this._navigateTo(activity.jumpUrl);
                        }
                    }
                    resolve();
                },
            });
        });
    }

    /** @param {number} d1 @param {number} d2 @returns {boolean} */
    _isSameDay(d1, d2) {
        const a = new Date(d1);
        const b = new Date(d2);
        return a.getFullYear() === b.getFullYear() &&
               a.getMonth() === b.getMonth() &&
               a.getDate() === b.getDate();
    }

    _navigateTo(url) {
        // 根据 URL 跳转到对应页面
        // 'activity://daily_task' → 每日任务页
        // 'activity://event?id=xxx' → 节日活动页
        // 'room://create' → 创建房间
    }
}
```

### 11.3 跑马灯组件

```javascript
/**
 * 跑马灯组件
 * 在大厅/房间等待页顶部或底部滚动展示后台配置的公告文案
 */
class MarqueeBanner {
    /**
     * @param {cc.Node} container - 挂载节点
     * @param {'top'|'bottom'} position
     * @param {object} config
     * @param {string} config.text      - 文案
     * @param {number} config.speed     - 滚动速度 (px/s)
     * @param {string} config.textColor - '#FFFFFF'
     * @param {string} config.bgColor   - '#00000088'
     */
    static init(container, position, config) {
        const label = new cc.Node();
        const labelComp = label.addComponent(cc.Label);
        labelComp.string = config.text;
        labelComp.fontSize = 24;
        labelComp.lineHeight = 36;
        labelComp.node.color = this._hexToColor(config.textColor);

        // 背景
        const bg = new cc.Node();
        const bgComp = bg.addComponent(cc.Graphics);
        bgComp.fillColor = this._hexToColor(config.bgColor);

        // 根据文字长度动态滚动
        const textWidth = labelComp.node.width;
        const screenWidth = cc.view.getVisibleSize().width;

        if (textWidth > screenWidth) {
            // 滚动模式
            this._startScroll(label, textWidth, screenWidth, config.speed);
        } else {
            // 静态模式
            label.position = cc.v2(0, 0);
        }

        container.addChild(bg);
        container.addChild(label);
    }

    static _startScroll(node, textWidth, screenWidth, speed) {
        let offsetX = screenWidth;
        const dt = 1 / 60;

        node.schedule(() => {
            offsetX -= speed * dt;
            if (offsetX < -textWidth) {
                offsetX = screenWidth; // 循环
            }
            node.x = offsetX;
        }, dt);
    }

    static _hexToColor(hex) {
        const r = parseInt(hex.slice(1, 3), 16);
        const g = parseInt(hex.slice(3, 5), 16);
        const b = parseInt(hex.slice(5, 7), 16);
        return cc.color(r, g, b);
    }

    /**
     * 从服务端拉取当前活跃公告
     */
    static async fetchActive() {
        const response = await fetch('/api/announcement/active?position=marquee&scene=lobby');
        const { announcements } = await response.json();
        // announcements: [{ text, speed, textColor, bgColor, position }]
        return announcements;
    }
}
```

前端启动时主动拉取最新配置表，支持后台热更新：

```javascript
/**
 * 配置表热更新管理器
 */
class ConfigHotReloadManager {
    constructor() {
        /** @type {string} 当前生效的配置版本号 */
        this.currentVersion = '';
    }

    /**
     * 启动时拉取最新配置
     */
    async init() {
        const localVersion = wx.getStorageSync('config_version') || '0.0.0';
        const response = await fetch('/api/config/version?local=' + localVersion);
        const { latestVersion, needUpdate, configFiles } = await response.json();

        if (needUpdate) {
            await this._applyUpdate(configFiles, latestVersion);
        }

        // 定时检查（每5分钟）
        setInterval(() => this._checkUpdate(), 5 * 60 * 1000);
    }

    async _checkUpdate() {
        const response = await fetch('/api/config/version?local=' + this.currentVersion);
        const { needUpdate, configFiles, latestVersion } = await response.json();
        if (needUpdate) {
            // 有新配置但不强制刷新 — 下次进入对局时生效
            wx.setStorageSync('pending_config_version', latestVersion);
            wx.setStorageSync('pending_config_files', configFiles);
            Logger.info('New config available, apply on next battle');
        }
    }

    async _applyUpdate(configFiles, version) {
        for (const file of configFiles) {
            const jsonData = await fetch(file.url).then(r => r.json());
            // 写入全局配置表，冻结防止篡改
            window[`CONFIG_${file.name}`] = Object.freeze(jsonData);
        }
        this.currentVersion = version;
        wx.setStorageSync('config_version', version);
        Logger.info(`Config updated to ${version}`);
    }

    /**
     * 对局开始前检查是否有待应用的配置
     */
    static async ensureLatestBeforeBattle() {
        const pendingVersion = wx.getStorageSync('pending_config_version');
        if (pendingVersion && pendingVersion !== ConfigHotReloadManager.instance.currentVersion) {
            const pendingFiles = wx.getStorageSync('pending_config_files');
            await ConfigHotReloadManager.instance._applyUpdate(pendingFiles, pendingVersion);
            wx.removeStorageSync('pending_config_version');
            wx.removeStorageSync('pending_config_files');
        }
    }
}
```

## 12. 每日任务与每周挑战 `🆕`

> 📌 来自需求 §12.1：日常运营活动，提升留存。

```javascript
/**
 * 每日任务/每周挑战管理器
 * 服务端根据配置下发任务，客户端渲染+追踪进度+领取奖励
 */
class DailyQuestManager {
    constructor() {
        /** @type {Quest[]} */
        this.dailyQuests = [];
        /** @type {Quest[]} */
        this.weeklyQuests = [];
    }

    /**
     * 拉取当前活跃任务
     */
    async loadQuests() {
        const response = await fetch('/api/quests/active');
        const { daily, weekly } = await response.json();
        this.dailyQuests = daily;
        this.weeklyQuests = weekly;
    }

    /**
     * 更新任务进度（由战斗事件触发）
     * @param {string} eventType - 'battle_complete'|'monster_kill'|'tower_place'|'ad_watch'
     * @param {object} eventData
     */
    async updateProgress(eventType, eventData) {
        const updated = [];

        for (const quest of [...this.dailyQuests, ...this.weeklyQuests]) {
            if (quest.completed) continue;

            let matched = false;
            switch (quest.condition.type) {
                case 'battle_count':
                    if (eventType === 'battle_complete') matched = true;
                    break;
                case 'monster_kill':
                    if (eventType === 'monster_kill') matched = true;
                    break;
                case 'use_specific_tower':
                    if (eventType === 'battle_complete' &&
                        eventData.towersUsed?.includes(quest.condition.towerType)) matched = true;
                    break;
                case 'watch_ad':
                    if (eventType === 'ad_watch') matched = true;
                    break;
            }

            if (matched) {
                quest.progress = Math.min(
                    (quest.progress || 0) + 1,
                    quest.condition.target
                );
                updated.push(quest);
            }
        }

        if (updated.length > 0) {
            await this._syncProgress(updated);
        }
    }

    /**
     * 领取奖励
     * @param {string} questId
     * @returns {boolean}
     */
    async claimReward(questId) {
        const quest = [...this.dailyQuests, ...this.weeklyQuests]
            .find(q => q.id === questId);
        if (!quest || quest.progress < quest.condition.target || quest.claimed) {
            return false;
        }

        const response = await fetch(`/api/quests/${questId}/claim`, { method: 'POST' });
        const { success, rewards } = await response.json();
        if (success) {
            quest.claimed = true;
            // 发放奖励
            for (const reward of rewards) {
                this._grantReward(reward);
            }
            Analytics.track('quest_claim', { questId, type: quest.type, rewards });
        }
        return success;
    }

    _grantReward(reward) {
        if (reward.type === 'gold') EconomyManager.instance.addGold(playerId, reward.amount);
        if (reward.type === 'diamond') UserDataManager.instance.addDiamond(reward.amount);
        if (reward.type === 'gacha') GachaSystem.instance.grantFreeDraw(reward.count);
    }

    /**
     * 每日0点刷新
     */
    onDayRefresh() {
        this.dailyQuests = [];
        this.loadQuests();
    }

    /**
     * 每周一0点刷新
     */
    onWeekRefresh() {
        this.dailyQuests = [];
        this.weeklyQuests = [];
        this.loadQuests();
    }

    async _syncProgress(quests) {
        await fetch('/api/quests/progress', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                updates: quests.map(q => ({ id: q.id, progress: q.progress })),
            }),
        });
    }
}

/**
 * @typedef {object} Quest
 * @property {string} id
 * @property {'daily'|'weekly'} type
 * @property {string} name             - 如「完成3场对局」
 * @property {string} description
 * @property {object} condition        - { type, target, [towerType] }
 * @property {number} progress         - 当前进度
 * @property {boolean} completed       - 是否完成
 * @property {boolean} claimed         - 是否已领取
 * @property {object[]} rewards        - [{ type, amount }]
 */
```

## 13. 数据同步策略 `🆕`

> 📌 来自需求 §12.4：云存档/数据同步，以 openid 为 key。

```javascript
/**
 * 云存档管理器
 * 以服务端为权威数据源，本地仅做缓存
 */
class CloudSyncManager {
    /**
     * 登录后同步数据
     * 策略：以服务端为准（Last-Write-Wins: Server）
     */
    static async syncOnLogin() {
        // 1. 拉取服务端用户数据
        const response = await fetch('/api/user/sync', {
            headers: { Authorization: `Bearer ${AuthManager.instance.token}` },
        });
        const serverData = await response.json();

        // 2. 比较本地缓存
        const localData = {
            talent: wx.getStorageSync('talent_data'),
            settings: wx.getStorageSync('settings_data'),
            achievements: wx.getStorageSync('achievement_data'),
            collection: wx.getStorageSync('collection_data'),
        };

        // 3. 服务端为准合并
        const merged = this._mergeServerWins(serverData, localData);

        // 4. 写入本地缓存
        wx.setStorageSync('talent_data', merged.talent);
        wx.setStorageSync('settings_data', merged.settings);
        wx.setStorageSync('achievement_data', merged.achievements);
        wx.setStorageSync('collection_data', merged.collection);
    }

    /**
     * 关键操作后同步到服务端
     * @param {string} key   - 'talent'|'settings'|'achievements'|'collection'
     * @param {object} data
     */
    static async syncToServer(key, data) {
        try {
            await fetch(`/api/user/sync/${key}`, {
                method: 'PUT',
                headers: {
                    'Content-Type': 'application/json',
                    Authorization: `Bearer ${AuthManager.instance.token}`,
                },
                body: JSON.stringify({ data }),
            });
        } catch (err) {
            // 离线时写入本地队列，联网后批量重试
            SyncQueue.enqueue(key, data);
            Logger.warn('Sync failed, queued for retry', { key });
        }
    }

    /**
     * 服务端为准合并
     * @returns {object}
     */
    static _mergeServerWins(serverData, localData) {
        // 天赋树：服务端为准
        // 成就进度：取较大值
        // 设置：新设备拉服务端，其他以本地为准
        return {
            talent: serverData.talent || localData.talent,
            achievements: this._mergeMaxProgress(
                serverData.achievements, localData.achievements
            ),
            collection: serverData.collection || localData.collection,
            settings: localData.settings || serverData.settings,
        };
    }

    /**
     * 成就进度取较大值
     */
    static _mergeMaxProgress(server, local) {
        if (!server && !local) return {};
        if (!server) return local;
        if (!local) return server;
        const merged = { ...server };
        for (const [id, val] of Object.entries(local)) {
            if ((merged[id] || 0) < val) merged[id] = val;
        }
        return merged;
    }
}

/**
 * 离线同步队列
 */
class SyncQueue {
    static _queue = [];

    static enqueue(key, data) {
        this._queue.push({ key, data, timestamp: Date.now() });
        wx.setStorageSync('sync_queue', this._queue);
    }

    static async flush() {
        const queue = wx.getStorageSync('sync_queue') || [];
        this._queue = queue;

        const failed = [];
        for (const item of this._queue) {
            try {
                await CloudSyncManager.syncToServer(item.key, item.data);
            } catch {
                failed.push(item);
            }
        }

        this._queue = failed;
        wx.setStorageSync('sync_queue', failed);
    }
}
```

**同步策略速查**：

| 数据类型 | 本地存储 | 服务端存储 | 冲突策略 |
|---------|:---:|:---:|---------|
| 天赋树配置 | wx.setStorage | MongoDB | **服务端为准** |
| 成就进度 | wx.setStorage | MongoDB | **取最大值** |
| 图鉴收集 | wx.setStorage | MongoDB | **服务端为准** |
| 用户设置 | wx.setStorage | MongoDB | **本地为准**（首次取服务端） |
| 对局记录 | — | MongoDB | **仅服务端** |
| 皮肤拥有 | — | MongoDB | **服务端为准** |

```javascript
class HapticManager {

        light: { label: '轻度', value: 'light' },

        this.currentLevel = saved;

        if (!this.LEVELS[level]) return;
        this.currentLevel = level;
        wx.setStorageSync('haptic_level', level);
    }

    /**
     * 轻触反馈（建造塔）
     */
    static light() {
        if (this.currentLevel === 'off') return;
        wx.vibrateShort({ type: 'light' });
    }

    /**
     * 中等反馈（骰子出结果 / 抽卡）
     */
    static medium() {
        if (this.currentLevel === 'off') return;
        if (this.currentLevel === 'light') {
            wx.vibrateShort({ type: 'light' });
            return;
        }
        wx.vibrateShort({ type: 'medium' });
    }

    /**
     * 强烈反馈（BOSS 警告 / 水晶受伤）
     */
    static heavy() {
        if (this.currentLevel === 'off') return;
        if (this.currentLevel === 'light') {
            wx.vibrateShort({ type: 'light' });
            return;
        }
        wx.vibrateLong({}); // 完整震动
    }
}
```

```javascript
/**
 * 通用搜索/筛选工具
 */
class SearchFilterUtils {
    /**
     * 模糊搜索（支持的模块：成就/图鉴/商城）
     * @param {string} keyword
     * @param {T[]} items
     * @param {string[]} searchFields - 搜索字段名
     * @returns {T[]}
     * @template T
     */
    static fuzzySearch(keyword, items, searchFields) {
        if (!keyword || !keyword.trim()) return items;
        const kw = keyword.trim().toLowerCase();
        return items.filter(item =>
            searchFields.some(field => {
                const value = item[field];
                return value && String(value).toLowerCase().includes(kw);
            })
        );
    }

    /**
     * 分类筛选
     * @param {T[]} items
     * @param {string} field
     * @param {string} value       - '' 表示"全部"
     * @returns {T[]}
     * @template T
     */
    static filterByField(items, field, value) {
        if (!value) return items;
        return items.filter(item => item[field] === value);
    }
}
```

```javascript
     * 成就列表筛选面板
     */
    static render(container) {
        // 筛选栏
        // ┌────────────────────────────────────────────┐
        // │ [全部分类 ▼] [全部状态 ▼] [🔍 搜索成就...] │
        // └────────────────────────────────────────────┘

     * 应用筛选
     * @param {Achievement[]} achievements
     * @param {{ category?: string, status?: string, keyword?: string }} filters
     * @returns {Achievement[]}

    static apply(achievements, filters) {

            result = result.filter(a => !a.completed);

        // 关键词搜索
        if (filters.keyword) {
            result = SearchFilterUtils.fuzzySearch(filters.keyword, result, ['name', 'description']);
        }

        return result;
    }
}
```
