## 1. 性能优化策略

### 1.1 对象池

帧同步游戏中，每帧大量创建/销毁实体（子弹、特效、飘字），使用对象池避免GC卡顿。

```javascript
/**
 * 通用对象池
 * @template T
 */
class ObjectPool {
    /**
     * @param {function():T} factory     - 创建函数
     * @param {function(T):void} reset   - 重置函数
     * @param {number} [preAllocate=0]   - 预分配数量
     */
    constructor(factory, reset, preAllocate) {
        /** @type {T[]} */
        this._pool = [];
        this._factory = factory;
        this._reset = reset;

        preAllocate = preAllocate || 0;
        for (let i = 0; i < preAllocate; i++) {
            this._pool.push(factory());
        }
    }

    /** @returns {T} */
    acquire() {
        if (this._pool.length > 0) return this._pool.pop();
        return this._factory();
    }

    /**
     * 归还对象
     * @param {T} obj
     */
    release(obj) {
        this._reset(obj);
        this._pool.push(obj);
    }

    /** @returns {number} */
    get availableCount() { return this._pool.length; }
}

// ===== 使用示例 =====

// 子弹对象池
const bulletPool = new ObjectPool(
    () => ({
        id: '', towerId: '', targetId: '',
        startX: 0, startY: 0, endX: 0, endY: 0,
        damage: FixedPoint.ZERO, damageType: DamageType.PHYSICAL,
        isActive: false, progress: 0, speed: 10,
    }),
    (b) => {
        b.id = ''; b.towerId = ''; b.targetId = '';
        b.isActive = false; b.progress = 0;
        b.damage = FixedPoint.ZERO;
    },
    300, // 预分配300个
);

// 伤害飘字对象池
const damagePopupPool = new ObjectPool(
    () => ({
        text: '', x: 0, y: 0,
        color: '#FFFFFF', size: 'normal',
        isActive: false, timer: 0, duration: 0.8,
        velocityY: -50,
    }),
    (dp) => { dp.isActive = false; dp.timer = 0; },
    100,
);
```

### 1.2 视口裁剪

```javascript
class ViewportCuller {
    /**
     * 只渲染当前视口内的实体
     * @param {Enemy[]|Tower[]} entities
     * @param {{ x: number, y: number, w: number, h: number }} viewport
     * @returns {Enemy[]|Tower[]}
     */
    static cull(entities, viewport) {
        const margin = 2; // 额外渲染视口边缘外2格
        return entities.filter(e => {
            const ex = e.currentX || e.gridX + 0.5;
            const ey = e.currentY || e.gridY + 0.5;
            return ex > viewport.x - margin &&
                   ex < viewport.x + viewport.w + margin &&
                   ey > viewport.y - margin &&
                   ey < viewport.y + viewport.h + margin;
        });
    }
}
```

| 策略 | 实现方式 | 收益 |

### 1.4 性能监控

```javascript
class PerformanceMonitor {
    constructor() {
        this.fpsHistory = [];
        this.frameTimeHistory = [];
        this.logicFrameTimeHistory = [];
        this.memoryHistory = [];
        this.isMonitoring = false;
    }

    start() {
        this.isMonitoring = true;
        this._tick();
    }

    _tick() {
        if (!this.isMonitoring) return;

        const now = performance.now();
        if (this._lastTime) {
            const dt = now - this._lastTime;
            this.frameTimeHistory.push(dt);
            if (this.frameTimeHistory.length > 60) this.frameTimeHistory.shift();
        }
        this._lastTime = now;

        // 内存（微信API）
        const memInfo = wx.getPerformance?.() || {};
        this.memoryHistory.push(memInfo.usedJSHeapSize || 0);

        // FPS
        this.fpsHistory.push(this._calculateFPS());
        if (this.fpsHistory.length > 60) this.fpsHistory.shift();

        // 低性能告警
        const avgFps = this._average(this.fpsHistory.slice(-30));
        if (avgFps < 30) {
            this._onLowPerformance();
        }

        requestAnimationFrame(() => this._tick());
    }

    /** @returns {number} 当前FPS */
    get currentFPS() {
        return Math.round(this._average(this.fpsHistory.slice(-10)));
    }

    /** @returns {number} 平均帧时间(ms) */
    get avgFrameTime() {
        return Math.round(this._average(this.frameTimeHistory.slice(-30)));
    }

    /** @returns {string} 'GREEN'|'YELLOW'|'RED' */
    get performanceLevel() {
        const fps = this.currentFPS;
        if (fps >= 55) return 'GREEN';
        if (fps >= 30) return 'YELLOW';
        return 'RED';
    }

    _onLowPerformance() {
        // 自动降级：减少粒子数、飘字数
        EffectSystem.instance.maxParticles = 20;
        DamagePopupController.instance.maxConcurrent = 10;
        Logger.warn('Low performance detected, quality reduced');
    }

    /** @param {number[]} arr @returns {number} */
    _average(arr) {
        if (arr.length === 0) return 0;
        return arr.reduce((a, b) => a + b, 0) / arr.length;
    }

    _calculateFPS() {
        const recent = this.frameTimeHistory.slice(-10);
        if (recent.length < 2) return 60;
        const avgDT = this._average(recent);
        return avgDT > 0 ? 1000 / avgDT : 60;
    }
}
```

### 1.5 弱网自适应

```javascript
class NetworkAdapter {
    /**
     * 根据网络延迟自动调整策略
     * @param {number} ping - 延迟(ms)
     */
    static adapt(ping) {
        if (ping < 50) {
            // 优秀：正常渲染60fps
            return { renderFps: 60, showEffects: true, predictionFrames: 3 };
        } else if (ping < 150) {
            // 一般：正常渲染
            return { renderFps: 60, showEffects: true, predictionFrames: 3 };
        } else if (ping < 300) {
            // 弱网：降低渲染帧率，保留逻辑帧
            return { renderFps: 30, showEffects: true, predictionFrames: 4 };
        } else {
            // 极弱网：暂停+提示
            PauseController.instance.networkPause();
            return { renderFps: 15, showEffects: false, predictionFrames: 5 };
        }
    }
}
```

### 2.1 服务端校验层

所有客户端操作必须通过服务端校验，防止外挂修改：

```javascript
class BattleValidator {
    /**
     * 校验建塔操作
     * @param {object} action
     * @param {PlayerState} player
     * @param {GameState} game
     * @returns {boolean}
     */
    static validatePlaceTower(action, player, game) {
        // 1. 金币校验
        const config = TOWER_CONFIG[action.towerType];
        if (!config) return false;
        if (player.gold < config.cost.build) return false;

        // 2. 位置校验
        const tile = game.map.tiles[action.gridY]?.[action.gridX];
        if (tile !== TileType.PLACEABLE) return false;

        // 3. 未被占用
        if (game.getTowerAt(action.gridX, action.gridY)) return false;

        // 4. 塔已解锁
        if (!player.unlockedTowers.includes(action.towerType)) return false;

        // 5. 建造上限（联机）
        const playerTowerCount = game.getPlayerTowerCount(player.id);
        if (game.multiplayerMaxTowers && playerTowerCount >= game.multiplayerMaxTowers) return false;

        return true;
    }

    /**
     * 校验升级操作
     */
    static validateUpgradeTower(action, player, game) {
        const tower = game.towers.find(t => t.id === action.towerId);
        if (!tower) return false;
        if (tower.ownerId !== player.id) return false;
        if (tower.level >= 3) return false;

        const upgradeCost = TOWER_CONFIG[tower.type].cost.upgrade[tower.level - 1];
        if (player.gold < upgradeCost) return false;

        return true;
    }

    /**
     * 校验出售操作
     */
    static validateSellTower(action, player, game) {
        const tower = game.towers.find(t => t.id === action.towerId);
        if (!tower) return false;
        if (tower.ownerId !== player.id) return false;
        return true;
    }

    /**
     * 校验金币赠送
     */
    static validateGiftGold(action, player, game) {
        if (action.targetPlayerId === player.id) return false;
        const targetPlayer = game.players.find(p => p.id === action.targetPlayerId);
        if (!targetPlayer) return false;
        if (player.gold < action.goldAmount) return false;
        // 30秒内只能向同一人赠送1次
        if (player.lastGiftTo && player.lastGiftTo[action.targetPlayerId]) {
            if (Date.now() - player.lastGiftTo[action.targetPlayerId] < 30000) return false;
        }
        return true;
    }
}
```

```javascript
/**
 * 服务端骰子服务
 * 使用 crypto.randomInt 真随机，结果记录日志
 */
class DiceService {
    /**
     * @param {string} playerId
     * @param {number} roomId
     * @returns {{ diceValue: number, buffs: object[] }}
     */
    roll(playerId, roomId) {
        // 真随机1-6
        const diceValue = crypto.randomInt(1, 7);

        // 天赋影响概率（好运分支→在服务端计算）
        const luckBonus = this._getLuckBonus(playerId);
        let adjustedValue = diceValue;
        if (luckBonus > 0 && diceValue > 3 && Math.random() < luckBonus) {
            adjustedValue = crypto.randomInt(1, 4); // 强制1-3
        }

        const result = this._pickBuffs(adjustedValue);

        // 记录审计日志
        this._auditLog(playerId, roomId, diceValue, adjustedValue, result);

        // 风控检测
        this._checkAnomaly(playerId);

        return result;
    }

    _getLuckBonus(playerId) {
        // 天赋：好运I/II/III → +5%/+10%/+15%
        const player = UserService.find(playerId);
        if (!player) return 0;
        if (player.talents.includes('LUCK_3')) return 0.15;
        if (player.talents.includes('LUCK_2')) return 0.10;
        if (player.talents.includes('LUCK_1')) return 0.05;
        return 0;
    }

    _auditLog(playerId, roomId, value, adjusted, result) {
        Logger.info('dice_roll', { playerId, roomId, value, adjusted, result });
    }

    /**
     * 风控检测：连续5次同一数值触发告警
     */
    _checkAnomaly(playerId) {
        const recentRolls = CacheService.get(`dice_history:${playerId}`) || [];
        recentRolls.push(Date.now());
        if (recentRolls.length > 10) recentRolls.shift();

        const values = recentRolls.map(() => 0); // 简化，实际取最近值
        const allSame = values.every(v => v === values[0]) && values.length >= 5;
        if (allSame) {
            AlertService.trigger('DICE_ANOMALY', { playerId, values });
        }

        CacheService.set(`dice_history:${playerId}`, recentRolls);
    }
}
```

### 4.1 切后台主动/被动区分实现 `🆕`

> 📌 来自第四轮 Q8：通过微信 API 区分主动/被动切后台。

```javascript
class AppLifecycleMonitor {
    constructor() {
        this.isInBackground = false;
        this.backgroundStartTime = 0;
        this.activeBackgroundCount = 0;  // 主动切后台次数
        this.passiveBackgroundCount = 0; // 被动切后台次数
        this.totalActiveBackgroundSec = 0;   // 主动累计秒数
        this.totalPassiveBackgroundSec = 0;  // 被动累计秒数
    }

    /**
     * 监听小程序前后台切换
     * 微信 API: wx.onAppShow / wx.onAppHide
     */
    init() {
        // 切到后台
        wx.onAppHide(() => {
            this.isInBackground = true;
            this.backgroundStartTime = Date.now();
        });

        // 切回前台
        wx.onAppShow((res) => {
            if (!this.isInBackground) return;
            this.isInBackground = false;

            const duration = (Date.now() - this.backgroundStartTime) / 1000;

            // 通过 scene 参数区分主动/被动切后台
            // scene=1089(微信首页)/其他→主动切后台；scene=1038(来电)/1158(系统通知)→被动
            const isActive = this._isActiveBackground(res.scene);

            if (isActive) {
                this.activeBackgroundCount++;
                this.totalActiveBackgroundSec += duration;
                this._applyPenaltyIfNeeded(duration);
            } else {
                this.passiveBackgroundCount++;
                this.totalPassiveBackgroundSec += duration;
                // 被动切后台仍触发暂停，但不累计惩罚
            }
        });
    }

    /**
     * 判断是否为主动切后台
     * @param {number} scene - wx.onAppShow 的 scene 参数
     * @returns {boolean}
     */
    _isActiveBackground(scene) {
        // 主动场景：手动切到桌面、微信主页、其他App
        const activeScenes = [1089, 1001, 1005, 1006]; // 微信首页/其他
        // 被动场景：来电(1038)、系统通知(1158)、微信消息弹窗
        const passiveScenes = [1038, 1158, 1007, 1022];

        if (passiveScenes.includes(scene)) return false;
        return true; // 默认视为主动
    }

    /**
     * 根据切后台时长触发不同策略
     * @param {number} duration - 秒
     */
    _applyPenaltyIfNeeded(duration) {
        if (duration < 10) {
            // 无惩罚
        } else if (duration < 30) {
            PauseController.instance.autoPause('player_away', duration);
        } else if (duration < 120) {
            PauseController.instance.autoPauseWithVote('player_away_long', duration);
        } else {
            ReconnectHandler.instance.onPlayerAbandon('background_timeout');
        }

        // 累计惩罚计数
        if (this.totalActiveBackgroundSec > 180) {
            // 对局结束时经验-50%
            GameWorld.instance.pendingExpPenalty = 0.5;
        }
    }

    /**
     * 获取对局内切后台统计（结算页展示）
     */
    getStats() {
        return {
            activeCount: this.activeBackgroundCount,
            passiveCount: this.passiveBackgroundCount,
            activeTotalSec: this.totalActiveBackgroundSec,
            passiveTotalSec: this.totalPassiveBackgroundSec,
        };
    }
}
```

**关键差异**：

| 类型 | 触发场景示例 | 是否暂停 | 是否计入惩罚 |
|------|---------|:---:|:---:|
| 主动切后台 | 按Home键、切到微信主页、手动打开其他App | ✅ | ✅ |
| 被动切后台 | 来电、系统通知弹窗、微信消息弹窗 | ✅ | ❌ |

### 6.2 关键埋点事件（统一前缀 `rtd_`）

```javascript
class Analytics {
    static EVENT_PREFIX = 'rtd_';

    /**
     * 上报埋点事件
     * @param {string} eventName - 不含前缀
     * @param {object} data
     */
    static track(eventName, data) {
        const payload = {
            event_name: this.EVENT_PREFIX + eventName,  // 统一前缀
            timestamp: Date.now(),
            player_id: AuthManager.instance.currentUser?.openid,
            session_id: SessionManager.instance.id,
            // 自动采集
            device_model: wx.getSystemInfoSync().model,
            wx_version: wx.getSystemInfoSync().version,
            network_type: 'unknown', // wx.getNetworkType()
            data,
        };

        // 战斗相关自动附加 room_id / wave_number / difficulty
        const battle = BattleManager.instance;
        if (battle && battle.isPlaying) {
            payload.room_id = battle.roomId;
            payload.wave_number = battle.currentWave;
            payload.difficulty = battle.difficulty;
        }

        // 批量上报（每10条或每30秒）
        AnalyticsBuffer.add(payload);
    }
}
```
